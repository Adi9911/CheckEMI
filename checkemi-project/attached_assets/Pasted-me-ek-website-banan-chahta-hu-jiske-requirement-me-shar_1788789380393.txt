### `utils/formulas.js`
// 100% CORRECT FORMULAS
export function calcEMI(P, annualRate, years){
  const r = annualRate/12/100; const n = years*12;
  if(r===0) return {emi:Math.round(P/n), totalInterest:0, totalPay:P};
  const emi = (P*r*Math.pow(1+r,n))/(Math.pow(1+r,n)-1);
  return {emi:Math.round(emi), totalPay:Math.round(emi*n), totalInterest:Math.round(emi*n-P)};
}
export function calcSIP(m, rate, years){
  const i = rate/12/100; const n = years*12;
  const fv = m * ((Math.pow(1+i,n)-1)/i)*(1+i);
  return {fv:Math.round(fv), invested:m*n, gain:Math.round(fv-m*n)};
}
export function calcPPF(y, years, rate=7.1){
  let b=0; for(let k=0;k<years;k++){ b=(b+y)*(1+rate/100); }
  return {maturity:Math.round(b), invested:y*years, interest:Math.round(b-y*years)};
}
export function calcFD(p, rate, years){
  const m = p*Math.pow(1+rate/100, years); return {maturity:Math.round(m), interest:Math.round(m-p)};
}
export function calcRD(m, rate, years){
  const i=rate/12/100; const n=years*12; let b=0;
  for(let k=0;k<n;k++){ b=(b+m)*(1+i); } // simplified RD
  // accurate formula:
  const fv = m * ((Math.pow(1+i,n)-1)/i)*(1+i);
  return {maturity:Math.round(fv), invested:m*n, interest:Math.round(fv-m*n)};
}
export function calcGST(amt, rate, inc=false){
  if(inc){ const base=amt/(1+rate/100); return {base:Math.round(base), gst:Math.round(amt-base), total:amt}; }
  const gst=amt*rate/100; return {base:amt, gst:Math.round(gst), total:Math.round(amt+gst)};
}
